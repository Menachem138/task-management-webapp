3. Proposed Changes
a) Search Logic Updates
```typescript
interface SearchResult {
  question: Question;
  score: number;
  matches: string[];
}

function calculateSearchScore(question: Question, keywords: string[]): SearchResult {
  const searchText = normalizeString(`${question.question} ${question.author} ${question.date}`);
  const matches = [];
  let score = 0;
  
  for (const keyword of keywords) {
    if (searchInText(searchText, keyword)) {
      score++;
      matches.push(keyword);
    }
  }
  
  return { question, score, matches };
}
```

b) Result Highlighting
```typescript
function highlightMatches(text: string, matches: string[]): string {
  let highlighted = text;
  for (const match of matches) {
    const regex = new RegExp(`(${escapeRegExp(match)})`, 'gi');
    highlighted = highlighted.replace(regex, '<mark>$1</mark>');
  }
  return highlighted;
}
```
